#include <iostream>

#include <string>
using namespace std;

struct Node {
    char ch;
    int freq;
    Node *left, *right;

    Node(char ch, int freq) {
        this->ch = ch;
        this->freq = freq;
        left = right = nullptr;
    }
};

struct Compare {
    bool operator()(Node *a, Node *b) {
        return a->freq > b->freq;
    }
};

void printCodes(Node *root, string code, int &totalBits) {
    if (root == nullptr)
        return;

    if (root->left == nullptr && root->right == nullptr) {
        if (code.empty())
            code = "0";

        cout << root->ch << "\t" << root->freq << "\t" << code << endl;

        totalBits += root->freq * code.length();
        return;
    }

    printCodes(root->left, code + "0", totalBits);
    printCodes(root->right, code + "1", totalBits);
}

    int n;

    cout << "Enter number of characters: ";

    if (n <= 0) {

        return 0;

    vector<char> chars(n);
    vector<int> freq(n);

    cout << "Enter each character and its frequency:\n";

    for (int i = 0; i < n; i++) {
        cin >> chars[i] >> freq[i];

        if (freq[i] <= 0) {
            cout << "Frequency must be positive.";
            return 0;
        }
    }

    priority_queue<Node *, vector<Node *>, Compare> pq;

    for (int i = 0; i < n; i++) {
        pq.push(new Node(chars[i], freq[i]));
    }

    while (pq.size() > 1) {
        Node *left = pq.top();
        pq.pop();

        Node *right = pq.top();
        pq.pop();

        Node *parent = new Node('\0', left->freq + right->freq);

        parent->left = left;
        parent->right = right;

        pq.push(parent);
    }

    Node *root = pq.top();

    int totalBits = 0;

    cout << "\nHuffman Codes:\n";
    cout << "Char\tFreq\tCode\n";

    printCodes(root, "", totalBits);

    cout << "\nTotal bits after encoding: " << totalBits << endl;

    return 0;
}

#include <iostream>
#include <queue>
#include <vector>
#include <string>
using namespace std;

// ==========================================
// 1. Huffman Tree Node Definition
// ==========================================
struct Node {
    char ch;        // Character symbol (leaf nodes store actual chars; internal nodes use '\0')
    int freq;       // Frequency (weight) of occurrence
    Node *left;     // Left child pointer (assigned binary path '0')
    Node *right;    // Right child pointer (assigned binary path '1')

    // Constructor to initialize a node with a character and frequency
    Node(char ch, int freq) {
        this->ch = ch;
        this->freq = freq;
        left = right = nullptr; // Initialize child pointers to null
    }
};

// ==========================================
// 2. Custom Comparator for Min-Heap
// ==========================================
// priority_queue in C++ is a max-heap by default.
// Overloading operator() with '>' converts it into a min-heap
// so the node with the lowest frequency stays at the top.
struct Compare {
    bool operator()(Node *a, Node *b) {
        return a->freq > b->freq;
    }
};

// ==========================================
// 3. Tree Traversal & Code Generation
// ==========================================
// Traverses the Huffman Tree to generate prefix-free variable-length codes.
// Appends '0' when traversing left and '1' when traversing right.
void printCodes(Node *root, string code, int &totalBits) {
    if (root == nullptr)
        return;

    // Base condition: Reached a leaf node (contains the actual character)
    if (root->left == nullptr && root->right == nullptr) {
        // Edge-case handler: If input consists of only 1 unique character,
        // the code string would remain empty, so assign "0" as a 1-bit code.
        if (code.empty())
            code = "0";

        // Print character, original frequency, and generated Huffman code
        cout << root->ch << "\t" << root->freq << "\t" << code << endl;

        // Accumulate total bits required: frequency * bit-length of code
        totalBits += root->freq * code.length();
        return;
    }

    // Traverse left subtree (append '0')
    printCodes(root->left, code + "0", totalBits);

    // Traverse right subtree (append '1')
    printCodes(root->right, code + "1", totalBits);
}

// ==========================================
// 4. Main Driver Program
// ==========================================
int main() {
    int n;

    // Read total number of unique characters
    cout << "Enter number of characters: ";
    cin >> n;

    // Input validation: Must have at least 1 character
    if (n <= 0) {
        cout << "Invalid number of characters.";
        return 0;
    }

    vector<char> chars(n);
    vector<int> freq(n);

    // Read character-frequency pairs
    cout << "Enter each character and its frequency:\n";
    for (int i = 0; i < n; i++) {
        cin >> chars[i] >> freq[i];

        // Input validation: Frequencies must be positive integers
        if (freq[i] <= 0) {
            cout << "Frequency must be positive.";
            return 0;
        }
    }

    // Min-Heap (priority queue) to efficiently fetch lowest frequency nodes in O(log k)
    priority_queue<Node *, vector<Node *>, Compare> pq;

    // Step 1: Create a leaf node for each character and insert into the min-heap
    for (int i = 0; i < n; i++) {
        pq.push(new Node(chars[i], freq[i]));
    }

    // Step 2: Build the Huffman Tree greedily
    // Repeat until only one combined node (the root) remains in the priority queue
    while (pq.size() > 1) {
        // Extract two lowest-frequency nodes
        Node *left = pq.top();
        pq.pop();

        Node *right = pq.top();
        pq.pop();

        // Create a new internal node with frequency equal to sum of two lowest nodes.
        // '\0' indicates this is an internal routing node, not an actual symbol.
        Node *parent = new Node('\0', left->freq + right->freq);

        parent->left = left;   // Link lowest frequency node to left
        parent->right = right; // Link second lowest frequency node to right

        // Push combined parent node back into the min-heap
        pq.push(parent);
    }

    // The remaining node is the root of the entire Huffman Tree
    Node *root = pq.top();

    int totalBits = 0;

    // Step 3: Traverse the tree to generate codes and calculate total compressed size
    cout << "\nHuffman Codes:\n";
    cout << "Char\tFreq\tCode\n";

    printCodes(root, "", totalBits);

    // Output overall bits required after Huffman encoding
    cout << "\nTotal bits after encoding: " << totalBits << endl;

    return 0;
}

# Practical: Implementation of Huffman Coding Using Greedy Algorithm

## 1. Aim

To write a C++ program to implement Huffman Coding using the Greedy Algorithm and generate optimal prefix codes for characters based on their frequencies to achieve lossless data compression.

## 2. Theory

### What is Huffman Coding?

Huffman Coding is a lossless data compression technique that reduces the number of bits required to represent data without losing any information.

It assigns variable-length binary codes to characters based on their frequencies of occurrence.

- Characters with higher frequencies receive shorter binary codes.
- Characters with lower frequencies receive longer binary codes.
- No code is a prefix of another code, which makes decoding unambiguous.

Example:

Consider the following characters and their frequencies:

| Character | Frequency |
| --------- | --------- |
| A         | 5         |
| B         | 9         |
| C         | 12        |
| D         | 13        |
| E         | 16        |
| F         | 45        |

In Huffman Coding, character F has the highest frequency, so it generally receives a shorter code than less frequent characters.

### 3. Greedy Algorithm Used in Huffman Coding

Huffman Coding uses a Greedy Algorithm to construct an optimal binary tree.

Greedy strategy:

At every step, select the two nodes having the minimum frequencies and combine them into a new parent node.

The process continues until only one node remains. This node becomes the root of the Huffman Tree.

### 4. Working of Huffman Coding

The algorithm follows these steps:

1. Read the characters and their corresponding frequencies.
2. Create a separate leaf node for each character.
3. Insert all nodes into a Min Priority Queue.
4. Extract the two nodes having the smallest frequencies.
5. Create a new parent node with the sum of both frequencies.
6. Assign the first extracted node as the left child and the second as the right child.
7. Insert the new parent node back into the priority queue.
8. Repeat until only one node remains.
9. Traverse the Huffman Tree to generate binary codes.
10. Assign 0 to every left edge and 1 to every right edge.
11. Calculate the total number of bits required for encoding.

## 8. Applications of Huffman Coding

1. File Compression: Used in compression formats such as ZIP and other compression systems.
2. Image Compression: Used as an entropy-coding technique in formats such as JPEG.
3. Data Transmission: Reduces the number of bits required to transmit information.

2. Assigns shorter codes to frequently occurring characters.
3. Reduces the total number of bits required for data representation.
4. Uses a greedy strategy to construct an optimal prefix code.
5. Produces uniquely decodable codes.

## 10. Disadvantages

1. Requires frequency information before encoding the data.
2. Requires additional storage for the Huffman Tree or code table.
3. Tree construction creates some processing overhead.
4. Compression efficiency depends on the frequency distribution of characters.
5. Less effective when all characters occur with nearly equal frequencies.
6. The compressed data requires decoding information to reconstruct the original data.
