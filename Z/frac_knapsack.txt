#include <iostream>
#include <algorithm>
#include <iomanip>
#include <vector>
using namespace std;

struct Item {
    int weight, profit;
    double ratio;
};

bool compare(Item a, Item b) {
    return a.ratio > b.ratio;
}

int main() {
    int n, capacity;
    double totalProfit = 0;

    cout << "Enter number of items: ";
    cin >> n;

    if (n <= 0) {
        cout << "Invalid number of items.";
        return 0;
    }

    vector<Item> items(n);

    for (int i = 0; i < n; i++) {
        cout << "Enter weight and profit of item "
             << i + 1 << ": ";
        cin >> items[i].weight >> items[i].profit;

        if (items[i].weight <= 0 || items[i].profit < 0) {
            cout << "Invalid weight or profit.";
            return 0;
        }

        items[i].ratio =
            (double)items[i].profit / items[i].weight;
    }

    cout << "Enter knapsack capacity: ";
    cin >> capacity;

    if (capacity < 0) {
        cout << "Invalid capacity.";
        return 0;
    }

    sort(items.begin(), items.end(), compare);

    cout << fixed << setprecision(2);
    cout << "\nSelected Items:\n";

    for (int i = 0; i < n; i++) {
        if (capacity == 0)
            break;

        if (items[i].weight <= capacity) {
            capacity -= items[i].weight;
            totalProfit += items[i].profit;

            cout << "Weight Taken: " << items[i].weight
                 << ", Profit: " << items[i].profit << "\n";
        }
        else {
            double fraction =
                (double)capacity / items[i].weight;

            double profitEarned =
                items[i].profit * fraction;

            cout << "Weight Taken: " << capacity
                 << ", Profit: " << profitEarned << "\n";

            totalProfit += profitEarned;
            capacity = 0;
        }
    }

    cout << "\nMaximum Profit: " << totalProfit << endl;

    return 0;
}

#include <iostream>

#include <iomanip>
#include <vector>
using namespace std;

// ==========================================
// 1. Item Structure Definition
// ==========================================
// Represents each item with its weight, profit, and profit-to-weight ratio
struct Item {
    int weight;     // Weight of the item
    int profit;     // Profit/value earned by taking the item
    double ratio;   // Profit per unit weight: profit / weight
};

// ==========================================
// 2. Custom Comparator for Greedy Strategy
// ==========================================
// Sorts items in descending order of profit-to-weight ratio.
// Higher ratio means more value per unit of weight capacity consumed.
bool compare(Item a, Item b) {
    return a.ratio > b.ratio;
}

int main() {
    int n, capacity;
    double totalProfit = 0; // Accumulates maximum profit (double to support fractional values)

    // Step 1: Input total number of items
    cout << "Enter number of items: ";
    cin >> n;

    // Input validation: Must have at least 1 item
    if (n <= 0) {
        cout << "Invalid number of items.";
        return 0;
    }

    vector<Item> items(n);

    // Step 2: Input weight and profit for each item
    for (int i = 0; i < n; i++) {
        cout << "Enter weight and profit of item "
             << i + 1 << ": ";
        cin >> items[i].weight >> items[i].profit;

        // Input validation: Weight must be positive, profit cannot be negative
        if (items[i].weight <= 0 || items[i].profit < 0) {
            cout << "Invalid weight or profit.";
            return 0;
        }

        // Calculate unit value (profit per unit weight) using explicit type casting
        items[i].ratio =
            (double)items[i].profit / items[i].weight;
    }

    // Step 3: Input the maximum weight capacity of the knapsack
    cout << "Enter knapsack capacity: ";
    cin >> capacity;

    // Input validation: Knapsack capacity cannot be negative
    if (capacity < 0) {
        cout << "Invalid capacity.";
        return 0;
    }

    // Step 4: GREEDY STEP - Sort items descending by profit/weight ratio
    // Runs in O(n log n) time
    sort(items.begin(), items.end(), compare);

    // Format floating-point outputs to 2 decimal places
    cout << fixed << setprecision(2);
    cout << "\nSelected Items:\n";

    // Step 5: KNAPSACK FILLING - Iterate through sorted items
    for (int i = 0; i < n; i++) {
        // If no capacity remains, stop early
        if (capacity == 0)
            break;

        // Case A: The whole item fits into the remaining capacity
        if (items[i].weight <= capacity) {
            capacity -= items[i].weight;      // Deduct item weight from knapsack capacity
            totalProfit += items[i].profit;   // Add entire item profit

            cout << "Weight Taken: " << items[i].weight
                 << ", Profit: " << items[i].profit << "\n";
        }
        // Case B: Item exceeds remaining capacity -> Take the optimal fraction
        else {
            // Fraction of item that fits into the remaining capacity
            double fraction =
                (double)capacity / items[i].weight;

            // Proportional profit earned from this fractional weight
            double profitEarned =
                items[i].profit * fraction;

            cout << "Weight Taken: " << capacity
                 << ", Profit: " << profitEarned << "\n";

            totalProfit += profitEarned; // Add fractional profit
            capacity = 0;                // Knapsack is now completely full
        }
    }

    // Step 6: Print final accumulated maximum profit
    cout << "\nMaximum Profit: " << totalProfit << endl;

    return 0;
}

# Practical: Fractional Knapsack Problem Using Greedy Algorithm

## 1. Aim

To write a C++ program to solve the Fractional Knapsack Problem using the Greedy Method and maximize the total profit by selecting items based on their profit-to-weight ratio.

## 2. Theory

### What is the Knapsack Problem?

The Knapsack Problem is an optimization problem in which a person has a bag (knapsack) with a limited weight capacity and a collection of items.

Each item has:

- Weight: The amount of space or capacity occupied by an item.
- Profit: The value earned by selecting an item.
- Capacity: The maximum weight that the knapsack can hold.

The objective is to select items in such a way that the total profit is maximized without exceeding the knapsack capacity.

### What is Fractional Knapsack?

Fractional Knapsack is a type of Knapsack Problem in which items can be divided into smaller fractions.

This means we can select:

- The complete item.
- A fraction of an item.
- Multiple items until the knapsack becomes full.

The main objective is to maximize the total profit within the given capacity.

### 3. Greedy Algorithm

The Fractional Knapsack Problem is solved using the Greedy Algorithm.

A Greedy Algorithm makes the best possible choice at each step to achieve an optimal solution.

Greedy strategy used:

Select items according to their profit-to-weight ratio in descending order.

The formula is:

\\[ Ratio=\frac{Profit}{Weight} \\]

- Items with higher ratios are selected first.
- If the complete item fits, take the entire item.
- If the item does not fit completely, take only the required fraction.
- Stop when the knapsack becomes full.

## 4. Difference Between 0/1 and Fractional Knapsack

| 0/1 Knapsack                               | Fractional Knapsack                      |
| ------------------------------------------ | ---------------------------------------- |
| Items cannot be divided.                   | Items can be divided into fractions.     |
| Complete item is selected or rejected.     | Partial items can be selected.           |
| Greedy method is not always optimal.       | Greedy method gives an optimal solution. |
| Commonly solved using Dynamic Programming. | Commonly solved using Greedy Algorithm.  |
| More complex optimization.                 | Comparatively simpler optimization.      |

## 6. Algorithm

1. Start the program.
2. Read the number of items.
3. Read the weight and profit of each item.
4. Calculate the profit-to-weight ratio for every item.
5. Read the knapsack capacity.
6. Sort all items in descending order of their ratios.
7. Select the item having the highest ratio.
8. If the complete item fits, add its weight and profit to the knapsack.
9. Otherwise, calculate the fraction that can fit and add the corresponding profit.
10. Repeat until the knapsack becomes full or all items are considered.
11. Display the maximum profit.
12. Stop the program.

## 3. Why Are Different Algorithms Used?

Fractional Knapsack – Greedy Algorithm

- Select the item with the highest profit-to-weight ratio.
- Take the complete item if possible.
- Otherwise, take the required fraction.
- This approach always gives an optimal solution for fractional knapsack.

0/1 Knapsack – Dynamic Programming

- Each item has only two choices: select (1) or reject (0).
- We consider different combinations of items.
- Dynamic Programming stores previously calculated results to avoid repeated calculations.
- Greedy selection based only on profit-to-weight ratio may give a suboptimal answer.
