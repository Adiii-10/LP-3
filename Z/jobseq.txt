#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Job {
    char id;
    int deadline;
    int profit;
};

bool cmp(Job a, Job b) {
    return a.profit > b.profit;
}

int main() {
    int n;

    cout << "Enter number of jobs: ";
    cin >> n;

    vector<Job> jobs(n);

    cout << "Enter job id, deadline, profit for each job:\n";

    for (int i = 0; i < n; i++)
        cin >> jobs[i].id >> jobs[i].deadline >> jobs[i].profit;

    sort(jobs.begin(), jobs.end(), cmp);

    int maxDeadline = 0;

    for (auto &j : jobs)
        maxDeadline = max(maxDeadline, j.deadline);

    vector<char> slot(maxDeadline + 1, '-');
    vector<bool> filled(maxDeadline + 1, false);

    int totalProfit = 0, count = 0;

    for (int i = 0; i < n; i++) {
        for (int t = jobs[i].deadline; t >= 1; t--) {
            if (!filled[t]) {
                filled[t] = true;
                slot[t] = jobs[i].id;
                totalProfit += jobs[i].profit;
                count++;
                break;
            }
        }
    }

    cout << "\nJob sequence: ";

    for (int t = 1; t <= maxDeadline; t++)
        if (filled[t])
            cout << slot[t] << " ";

    cout << "\nJobs scheduled: " << count;
    cout << "\nTotal profit: " << totalProfit << endl;

    return 0;
}

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

// Structure to encapsulate the properties of a single job
struct Job {
    char id;       // Unique identifier for the job (e.g., 'a', 'b', 'J1')
    int deadline;  // Latest unit time slot by which the job must be completed
    int profit;    // Profit earned if the job is completed on or before deadline
};

// Comparator function used to sort jobs in strictly non-increasing (descending) order of profit
bool cmp(Job a, Job b) {
    return a.profit > b.profit; // Places higher-profit jobs first
}

int main() {
    int n;

    // 1. Read the total count of available jobs
    cout << "Enter number of jobs: ";
    cin >> n;

    vector<Job> jobs(n);

    // 2. Read each job's specifications
    cout << "Enter job id, deadline, profit for each job:\n";
    for (int i = 0; i < n; i++)
        cin >> jobs[i].id >> jobs[i].deadline >> jobs[i].profit;

    // 3. GREEDY STEP: Sort jobs descending by profit
    // Maximizing overall profit means considering the most lucrative jobs first
    sort(jobs.begin(), jobs.end(), cmp);

    // 4. Determine the size of the schedule timeline
    // The maximum possible time slot we can schedule into is the highest deadline among all jobs
    int maxDeadline = 0;
    for (auto &j : jobs)
        maxDeadline = max(maxDeadline, j.deadline);

    // 5. Initialize time slots (1-indexed, using indices 1 to maxDeadline)
    // slot: stores which job is placed in which time slot
    // filled: boolean flag indicating if a time slot is already taken
    vector<char> slot(maxDeadline + 1, '-');
    vector<bool> filled(maxDeadline + 1, false);

    int totalProfit = 0, count = 0;

    // 6. ALLOCATION STEP: Assign slots to sorted jobs greedily
    for (int i = 0; i < n; i++) {
        // Find a free slot for jobs[i], scanning backwards from its deadline down to time slot 1.
        // Reason: Scheduling a job as late as possible preserves earlier slots for jobs with tighter deadlines.
        for (int t = jobs[i].deadline; t >= 1; t--) {
            if (!filled[t]) {
                filled[t] = true;              // Mark slot 't' as occupied
                slot[t] = jobs[i].id;          // Assign current job to slot 't'
                totalProfit += jobs[i].profit; // Add profit to running total
                count++;                       // Increment count of scheduled jobs
                break;                         // Job scheduled; move to next job
            }
        }
    }

    // 7. Output the generated optimal schedule
    cout << "\nJob sequence: ";
    for (int t = 1; t <= maxDeadline; t++)
        if (filled[t])
            cout << slot[t] << " ";

    cout << "\nJobs scheduled: " << count;
    cout << "\nTotal profit: " << totalProfit << endl;

    return 0;
}

# Practical: Job Sequencing with Deadlines Using Greedy Method

## 1. Aim

To write a C++ program to solve the Job Sequencing with Deadlines problem using the Greedy Algorithm and maximize the total profit by scheduling jobs within their deadlines.

## 2. Theory

### What is Job Sequencing with Deadlines?

Job Sequencing with Deadlines is a scheduling problem in which a set of jobs is given, and each job has:

1. Job ID: A unique identifier used to identify each job.
2. Deadline: The maximum time by which a job must be completed.
3. Profit: The amount of profit earned after completing a job.

The objective is to schedule jobs in such a way that the maximum total profit is obtained without missing any deadlines.

In this problem, each job requires exactly one unit of time, and only one job can be performed at a time.

### 3. Greedy Algorithm

A Greedy Algorithm is an algorithmic technique that makes the best possible choice at each step to achieve an optimal solution.

In Job Sequencing, the greedy strategy is:

- Sort all jobs in decreasing order of their profits.
- Select the job with the highest profit first.
- Schedule the selected job in the latest available time slot before its deadline.
- If no available slot exists, reject that job.
- Repeat until all jobs are considered.

Why do we schedule jobs in the latest available slot?

Scheduling a job as late as possible before its deadline keeps earlier time slots available for other jobs having smaller deadlines.

## 5. Algorithm

1. Start the program.
2. Read the number of jobs and their IDs, deadlines and profits.
3. Sort all jobs in descending order of profit.
4. Find the maximum deadline among all jobs.
5. Create time slots and mark all slots as empty.
6. Select each job in decreasing order of profit.
7. Check the available time slots from its deadline towards slot 1.
8. If an empty slot is found, assign the job to that slot and add its profit.
9. If no empty slot is available, reject the job.
10. Display the scheduled jobs, number of jobs and total profit.
11. Stop the program.

### Working of Important Functions

1\. Sort Function

```
bool cmp(Job a, Job b) {
    return a.profit > b.profit;
}
```

This function sorts jobs in descending order of profit so that the most profitable job is considered first.

2\. Finding Maximum Deadline

```
for (auto &j : jobs)
    maxDeadline = max(maxDeadline, j.deadline);
```

This loop finds the largest deadline to determine the number of available time slots.

3\. Checking Available Slots

```
for (int t = jobs[i].deadline; t >= 1; t--) {
    if (!filled[t]) {
        filled[t] = true;
        slot[t] = jobs[i].id;
        totalProfit += jobs[i].profit;
        count++;
        break;
    }
}
```

This loop searches for an empty time slot starting from the job's deadline and moving backwards. If a slot is available, the job is scheduled and its profit is added.

## 7. Time and Space Complexity

### Time Complexity

Let:

- n = Number of jobs.
- D = Maximum deadline.

| Operation                | Complexity |
| ------------------------ | ---------- |
| Sorting jobs             | O(n log n) |
| Finding maximum deadline | O(n)       |
| Initializing slots       | O(D)       |
| Scheduling jobs          | O(n × D)   |

Overall Time Complexity:

\\[ O(n\log n+nD) \\]

The scheduling process checks at most D slots for each of n jobs.

Worst-case Time Complexity:

\\[ O(n^2) \\]

When D is proportional to n, the overall complexity becomes O(n²).

### Space Complexity

- Jobs vector requires O(n) space.
- Slot vector requires O(D) space.
- Filled vector requires O(D) space.

Overall Space Complexity:

\\[ O(n+D) \\]

1\. What is a Greedy Algorithm?

A Greedy Algorithm makes the locally optimal choice at each step to obtain an overall solution.

2\. What is Job Sequencing with Deadlines?

To maximize the total profit by scheduling the maximum possible profitable jobs within their deadlines.

To consider the most profitable jobs first and maximize the total profit.

O(n log n + nD), where n is the number of jobs and D is the maximum deadline. In the worst case, it is O(n²) when D is proportional to n.
