
const readline = require('readline');

const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

class BusinessNetwork {
    constructor() {
        this.products = [];
    }

    createProduct(id, name, owner, price) {
        this.products.push({ id, name, owner, price });
        console.log("\nProduct added successfully!");
    }

    displayProducts() {
        console.log("\n--- Business Network Products ---");

        if (this.products.length === 0) {
            console.log("No products available.");
            return;
        }

        this.products.forEach(p => {
            console.log("Product ID:", p.id);
            console.log("Product Name:", p.name);
            console.log("Owner:", p.owner);
            console.log("Price:", p.price);
            console.log("----------------------");
        });
    }

    transferProduct(id, newOwner) {
        const product = this.products.find(p => p.id === id);

        if (product) {
            product.owner = newOwner;
            console.log("\nOwnership transferred successfully!");
        } else {
            console.log("\nProduct not found!");
        }
    }
}

const network = new BusinessNetwork();

function menu() {
    console.log("\n===== BUSINESS NETWORK =====");
    console.log("1. Add Product");
    console.log("2. Display Products");
    console.log("3. Transfer Ownership");
    console.log("4. Exit");

    rl.question("Enter your choice: ", (choice) => {

        switch (choice) {
            case '1':
                rl.question("Enter Product ID: ", (id) => {
                    rl.question("Enter Product Name: ", (name) => {
                        rl.question("Enter Company Name: ", (owner) => {
                            rl.question("Enter Product Price: ", (price) => {

                                const productId = Number(id);
                                const productPrice = Number(price);

                                if (!Number.isInteger(productId) ||
                                    !Number.isFinite(productPrice) ||
                                    productId <= 0 || productPrice < 0 ||
                                    network.products.some(p => p.id === productId)) {
                                    console.log("Invalid input or duplicate Product ID!");
                                } else {
                                    network.createProduct(
                                        productId, name, owner, productPrice
                                    );
                                }

                                menu();
                            });
                        });
                    });
                });
                break;

            case '2':
                network.displayProducts();
                menu();
                break;

            case '3':
                rl.question("Enter Product ID: ", (id) => {
                    rl.question("Enter New Owner Name: ", (owner) => {
                        network.transferProduct(Number(id), owner);
                        menu();
                    });
                });
                break;

            case '4':
                console.log("Exiting Business Network...");
                rl.close();
                break;

            default:
                console.log("Invalid choice! Try again.");
                menu();
        }
    });
}

console.log("Welcome to Hyperledger Business Network");
menu();
